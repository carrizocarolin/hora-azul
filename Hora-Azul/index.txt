  <script>
    const padCount = 8;
    const padsEl = document.querySelector('#pads');
    const sequencerEl = document.querySelector('#sequencer');
    const initBtn = document.querySelector('#initBtn');
    const recordBtn = document.querySelector('#recordBtn');
    const stopBtn = document.querySelector('#stopBtn');
    const assignBtn = document.querySelector('#assignBtn');
    const statusText = document.querySelector('#statusText');
    const recStatus = document.querySelector('#recStatus');
    const playMode = document.querySelector('#playMode');
    const reverseBtn = document.querySelector('#reverseBtn');
    const selectedPadText = document.querySelector('#selectedPadText');
    const clearPadBtn = document.querySelector('#clearPadBtn');
    const downloadBtn = document.querySelector('#downloadBtn');
    const seqPlayBtn = document.querySelector('#seqPlayBtn');
    const seqStopBtn = document.querySelector('#seqStopBtn');
    const bpmInput = document.querySelector('#bpm');

    const volume = document.querySelector('#volume');
    const filter = document.querySelector('#filter');
    const q = document.querySelector('#q');
    const volVal = document.querySelector('#volVal');
    const filterVal = document.querySelector('#filterVal');
    const qVal = document.querySelector('#qVal');

    let audioCtx;
    let masterGain;
    let masterFilter;
    let micStream;
    let recorder;
    let chunks = [];
    let lastBlob = null;
    let selectedPad = 0;
    let activeSources = new Map();
    let seqTimer = null;
    let currentStep = 0;

    const pads = Array.from({ length: padCount }, (_, i) => ({
      name: `Pad ${i + 1}`,
      buffer: null,
      reversed: false,
      mode: 'oneshot'
    }));

    const pattern = Array.from({ length: padCount }, () => Array(16).fill(false));

    function setStatus(text) {
      statusText.textContent = text;
    }

    function createAudioGraph() {
      audioCtx = new AudioContext();
      masterGain = audioCtx.createGain();
      masterFilter = audioCtx.createBiquadFilter();
      masterFilter.type = 'lowpass';
      masterFilter.frequency.value = Number(filter.value);
      masterFilter.Q.value = Number(q.value);
      masterGain.gain.value = Number(volume.value);
      masterFilter.connect(masterGain).connect(audioCtx.destination);
    }

    async function initAudio() {
      if (!audioCtx) createAudioGraph();
      await audioCtx.resume();
      micStream = await navigator.mediaDevices.getUserMedia({ audio: true });
      recordBtn.disabled = false;
      initBtn.disabled = true;
      setStatus('Audio activo. Listo para grabar.');
    }

    function renderPads() {
      padsEl.innerHTML = '';
      pads.forEach((pad, i) => {
        const btn = document.createElement('button');
        btn.className = 'pad' + (i === selectedPad ? ' active' : '');
        btn.innerHTML = `<div class="num">${i + 1}</div><div class="name">${pad.buffer ? pad.name : 'Vacío'}</div><div class="mode">${pad.reversed ? 'rev · ' : ''}${pad.mode}</div>`;
        btn.addEventListener('mousedown', () => {
          selectedPad = i;
          selectedPadText.textContent = i + 1;
          playMode.value = pads[i].mode;
          renderPads();
          triggerPad(i);
        });
        btn.addEventListener('mouseup', () => stopPad(i));
        btn.addEventListener('mouseleave', () => { if (pads[i].mode === 'gate') stopPad(i); });
        padsEl.appendChild(btn);
      });
    }

    function renderSequencer() {
      sequencerEl.innerHTML = '';
      pattern.forEach((row, padIndex) => {
        const label = document.createElement('div');
        label.className = 'status';
        label.style.marginTop = '12px';
        label.textContent = `Pad ${padIndex + 1}`;
        sequencerEl.appendChild(label);

        const steps = document.createElement('div');
        steps.className = 'steps';
        row.forEach((isOn, stepIndex) => {
          const btn = document.createElement('button');
          btn.className = 'step' + (isOn ? ' on' : '') + (stepIndex === currentStep ? ' playhead' : '');
          btn.addEventListener('click', () => {
            pattern[padIndex][stepIndex] = !pattern[padIndex][stepIndex];
            renderSequencer();
          });
          steps.appendChild(btn);
        });
        sequencerEl.appendChild(steps);
      });
    }

    function makePlayableBuffer(buffer, reversed) {
      if (!reversed) return buffer;
      const rev = audioCtx.createBuffer(buffer.numberOfChannels, buffer.length, buffer.sampleRate);
      for (let ch = 0; ch < buffer.numberOfChannels; ch++) {
        const data = buffer.getChannelData(ch);
        const out = rev.getChannelData(ch);
        for (let i = 0; i < data.length; i++) out[i] = data[data.length - 1 - i];
      }
      return rev;
    }

    function triggerPad(index, when = 0) {
      const pad = pads[index];
      if (!audioCtx || !pad.buffer) return;

      if (pad.mode !== 'loop') stopPad(index);

      const source = audioCtx.createBufferSource();
      source.buffer = makePlayableBuffer(pad.buffer, pad.reversed);
      source.loop = pad.mode === 'loop';
      source.connect(masterFilter);
      source.start(when ? when : audioCtx.currentTime);

      if (!activeSources.has(index)) activeSources.set(index, []);
      activeSources.get(index).push(source);
      source.onended = () => {
        const list = activeSources.get(index) || [];
        activeSources.set(index, list.filter(s => s !== source));
      };
    }

    function stopPad(index) {
      const pad = pads[index];
      if (pad.mode !== 'gate' && pad.mode !== 'loop') return;
      const sources = activeSources.get(index) || [];
      sources.forEach(src => { try { src.stop(); } catch (_) {} });
      activeSources.set(index, []);
    }

    async function decodeBlobToBuffer(blob) {
      const arrayBuffer = await blob.arrayBuffer();
      return await audioCtx.decodeAudioData(arrayBuffer);
    }

    initBtn.addEventListener('click', async () => {
      try { await initAudio(); }
      catch (err) { setStatus('No pude acceder al micrófono. Revisá permisos del navegador.'); }
    });

    recordBtn.addEventListener('click', () => {
      chunks = [];
      recorder = new MediaRecorder(micStream);
      recorder.ondataavailable = e => chunks.push(e.data);
      recorder.onstop = () => {
        lastBlob = new Blob(chunks, { type: recorder.mimeType || 'audio/webm' });
        assignBtn.disabled = false;
        downloadBtn.disabled = false;
        recStatus.classList.remove('recording');
        setStatus('Grabación lista. Asignala a un pad.');
      };
      recorder.start();
      recStatus.classList.add('recording');
      recordBtn.disabled = true;
      stopBtn.disabled = false;
      setStatus('Grabando...');
    });

    stopBtn.addEventListener('click', () => {
      if (recorder && recorder.state === 'recording') recorder.stop();
      recordBtn.disabled = false;
      stopBtn.disabled = true;
    });

    assignBtn.addEventListener('click', async () => {
      if (!lastBlob) return;
      const buffer = await decodeBlobToBuffer(lastBlob);
      pads[selectedPad].buffer = buffer;
      pads[selectedPad].name = `Grabación ${new Date().toLocaleTimeString()}`;
      pads[selectedPad].mode = playMode.value;
      renderPads();
      setStatus(`Sample asignado al pad ${selectedPad + 1}.`);
    });

    playMode.addEventListener('change', () => {
      pads[selectedPad].mode = playMode.value;
      renderPads();
    });

    reverseBtn.addEventListener('click', () => {
      pads[selectedPad].reversed = !pads[selectedPad].reversed;
      renderPads();
    });

    clearPadBtn.addEventListener('click', () => {
      stopPad(selectedPad);
      pads[selectedPad] = { name: `Pad ${selectedPad + 1}`, buffer: null, reversed: false, mode: 'oneshot' };
      pattern[selectedPad].fill(false);
      renderPads();
      renderSequencer();
    });

    downloadBtn.disabled = true;
    downloadBtn.addEventListener('click', () => {
      if (!lastBlob) return;
      const url = URL.createObjectURL(lastBlob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'sample.webm';
      a.click();
      URL.revokeObjectURL(url);
    });

    volume.addEventListener('input', () => {
      volVal.textContent = volume.value;
      if (masterGain) masterGain.gain.value = Number(volume.value);
    });

    filter.addEventListener('input', () => {
      filterVal.textContent = `${filter.value} Hz`;
      if (masterFilter) masterFilter.frequency.value = Number(filter.value);
    });

    q.addEventListener('input', () => {
      qVal.textContent = q.value;
      if (masterFilter) masterFilter.Q.value = Number(q.value);
    });

    seqPlayBtn.addEventListener('click', async () => {
      if (!audioCtx) await initAudio();
      clearInterval(seqTimer);
      currentStep = 0;
      const tick = () => {
        const stepTime = audioCtx.currentTime + 0.015;
        pattern.forEach((row, padIndex) => {
          if (row[currentStep]) triggerPad(padIndex, stepTime);
        });
        renderSequencer();
        currentStep = (currentStep + 1) % 16;
      };
      tick();
      const intervalMs = (60 / Number(bpmInput.value)) * 1000 / 4;
      seqTimer = setInterval(tick, intervalMs);
    });

    seqStopBtn.addEventListener('click', () => {
      clearInterval(seqTimer);
      seqTimer = null;
      currentStep = 0;
      renderSequencer();
    });

    window.addEventListener('keydown', e => {
      const n = Number(e.key);
      if (n >= 1 && n <= padCount && !e.repeat) triggerPad(n - 1);
    });

    window.addEventListener('keyup', e => {
      const n = Number(e.key);
      if (n >= 1 && n <= padCount) stopPad(n - 1);
    });

    renderPads();
    renderSequencer();
  </script>
